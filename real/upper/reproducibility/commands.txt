# Run from the reproducibility directory, using a Python with python-flint.
# These commands do not install software, connect remotely, or submit jobs.

# Short complete verification of the supplied certificate:
python -B -u verify.py --output /tmp/real-upper-verification

# Prepare a separate new calculation without doing numerical work:
python -B -u reproduce.py prepare --workdir /scratch/real-upper-reproduction

# Complete reproduction on already assigned computational resources:
python -B -u reproduce.py full --workdir /scratch/real-upper-reproduction --workers 2

# Individual checkpointable phases:
python -B -u reproduce.py init --workdir /scratch/real-upper-reproduction
python -B -u reproduce.py block --workdir /scratch/real-upper-reproduction --block 0
# Repeat the preceding command for every integer block index 1 through 31.
python -B -u reproduce.py merge --workdir /scratch/real-upper-reproduction
python -B -u reproduce.py head --workdir /scratch/real-upper-reproduction
python -B -u reproduce.py norm --workdir /scratch/real-upper-reproduction
python -B -u reproduce.py final --workdir /scratch/real-upper-reproduction

# On assigned hardware without SLURM, an explicit override is necessary:
python -B -u reproduce.py full --workdir /tmp/real-upper-full --workers 1 --allow-long-local-computation
